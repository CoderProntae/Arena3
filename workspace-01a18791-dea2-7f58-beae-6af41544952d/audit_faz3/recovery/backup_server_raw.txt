   393→local function outwardFrontWave(first,last,seconds,color)
   394→	local wave={}
   395→	for i=0,35 do
   396→		local a=i*TAU/36;local b=(i+1)*TAU/36
   397→		wave[#wave+1]=beam(transient,"WINDOW_FACING_SHOCK_RING",
   398→			CORE+Vector3.new(math.cos(a)*first,math.sin(a)*first,29),
   399→			CORE+Vector3.new(math.cos(b)*first,math.sin(b)*first,29),
   400→			1.05,color or C.violet,Enum.Material.Neon,0.48)
   401→	end
   402→	task.spawn(function()
   403→		local started=now()
   404→		while now()-started<seconds do
   405→			local u=math.clamp((now()-started)/seconds,0,1)
   406→			local r=first+(last-first)*u
   407→			for i,p in ipairs(wave) do
   408→				local a=(i-1)*TAU/36;local b=i*TAU/36
   409→				local x=CORE+Vector3.new(math.cos(a)*r,math.sin(a)*r,29)
   410→				local y=CORE+Vector3.new(math.cos(b)*r,math.sin(b)*r,29)
   411→				p.CFrame=CFrame.lookAt((x+y)/2,y)
   412→				p.Size=UNITS.new(1.05*WORLD_SCALE,1.05*WORLD_SCALE,
   413→					(y-x).Magnitude)
   414→				p.Transparency=0.48+0.51*u
   415→			end

  2988→		state.temp=math.max(26,state.temp-0.31*dt)
  2989→		state.pressure=math.max(15,state.pressure-0.16*dt)
  2990→		return

  2995→		(state.lasers and 650 or 0)-coolant*4.5
