   153→local function buildEditableTexture(size, pixelFn)
   154→	local ei = keep(AssetService:CreateEditableImage({Size = Vector2.new(size, size)}))
   155→	local buf = buffer.create(size * size * 4)
   156→	local o = 0
   157→	for py = 0, size - 1 do
   158→		local v = (py + 0.5) / size
   159→		for px = 0, size - 1 do
   160→			local u = (px + 0.5) / size
   161→			local r, g, b, a = pixelFn(u, v)
   162→			buffer.writeu8(buf, o,     math.clamp(math.floor(r * 255 + 0.5), 0, 255))
   163→			buffer.writeu8(buf, o + 1, math.clamp(math.floor(g * 255 + 0.5), 0, 255))
   164→			buffer.writeu8(buf, o + 2, math.clamp(math.floor(b * 255 + 0.5), 0, 255))
   165→			buffer.writeu8(buf, o + 3, math.clamp(math.floor(a * 255 + 0.5), 0, 255))
   166→			o += 4
   167→		end
   168→	end
   169→	ei:WritePixelsBuffer(Vector2.zero, Vector2.new(size, size), buf)
   170→	return {content = Content.fromObject(ei), ref = ei}
   171→end

   408→-- Straight Axisymmetric High-Poly Conical 3D Funnel (24 x 48 = 2,304 triangles, 7.5° smooth circular perimeter!)
   409→local function buildAxisymmetricConeMesh(seed)
   410→	local em = keep(AssetService:CreateEditableMesh())
   411→	local rings, segs = 24, 48
   412→	local vids, nids, uvids = {}, {}, {}
   413→	for r = 0, rings do
   414→		local t = r / rings
   415→		vids[r] = {}; nids[r] = {}; uvids[r] = {}
   416→		local baseR = 0.16 + 0.34 * (t ^ 0.92)
   417→		for s = 0, segs do
   418→			local u = s / segs
   419→			local th = u * math.pi * 2
   420→			local c, sn = math.cos(th), math.sin(th)
   421→			local flute = math.sin(th * 5 + t * 5.2) * 0.032 * t
   422→			local rip = fbm3(c * 1.9 + seed, sn * 1.9, t * 2.8) * 0.042 * t
   423→			local rad = math.max(0.06, baseR + flute + rip)
   424→			local pos = Vector3.new(c * rad, sn * rad, -t)
   425→			vids[r][s] = em:AddVertex(pos)
   426→			nids[r][s] = em:AddNormal(Vector3.new(c * 0.88, sn * 0.88, 0.47).Unit)
   427→			uvids[r][s] = em:AddUV(Vector2.new(u, t))
   428→		end
   429→	end
   430→	for r = 0, rings - 1 do
   431→		for s = 0, segs - 1 do
   432→			local f1 = em:AddTriangle(vids[r][s], vids[r + 1][s], vids[r + 1][s + 1])
   433→			em:SetFaceNormals(f1, {nids[r][s], nids[r + 1][s], nids[r + 1][s + 1]})
   434→			em:SetFaceUVs(f1, {uvids[r][s], uvids[r + 1][s], uvids[r + 1][s + 1]})
   435→			local f2 = em:AddTriangle(vids[r][s], vids[r + 1][s + 1], vids[r][s + 1])
   436→			em:SetFaceNormals(f2, {nids[r][s], nids[r + 1][s + 1], nids[r][s + 1]})
   437→			em:SetFaceUVs(f2, {uvids[r][s], uvids[r + 1][s + 1], uvids[r][s + 1]})
   438→		end
   439→	end
   440→	return {em = em, content = toStaticMeshContent(em), tris = rings * segs * 2}
   441→end

   443→-- High-Poly Helical Cylindrical Laser Plasma Envelope (26 x 48 = 2,496 triangles, 7.5° smooth circular perimeter!)
   444→local function buildCylindricalLaserMesh()
   445→	local em = keep(AssetService:CreateEditableMesh())
   446→	local rings, segs = 26, 48
   447→	local vids, nids, uvids = {}, {}, {}

   449→		local t = r / rings
   450→		vids[r] = {}; nids[r] = {}; uvids[r] = {}

   459→			local rad = flare * (1 + w1)
   460→			vids[r][s] = em:AddVertex(Vector3.new(c * rad, sn * rad, z))
   461→			nids[r][s] = em:AddNormal(Vector3.new(c * 0.92, sn * 0.92, 0.39).Unit)

   467→			local f1 = em:AddTriangle(vids[r][s], vids[r + 1][s], vids[r + 1][s + 1])

   478→-- Ultra-Smooth High-Poly 3D Solar Spheres (44x76 = 6,688 tris & 36x64 = 4,608 tris -> ZERO polygonal facets!)
   479→local function buildSolar3DSphere(rings, segs, amp, freq, seed)
   480→	local em = keep(AssetService:CreateEditableMesh())

   482→	for r = 0, rings do

   497→			nids[r][s] = em:AddNormal(-dir)

   502→		for s = 0, segs - 1 do
   503→			local f1 = em:AddTriangle(vids[r][s], vids[r + 1][s + 1], vids[r + 1][s])
